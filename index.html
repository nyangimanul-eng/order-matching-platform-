async function requestWithdrawal() { const amount = Number( document .getElementById("withdrawAmount") .value ); const phone = document .getElementById("withdrawPhone") .value.trim(); if ( !Number.isInteger(amount) || amount <= 0 ) { showWithdrawMessage( "Enter a valid withdrawal amount.", true ); return; } if (!phone) { showWithdrawMessage( "Enter your M-PESA phone number.", true ); return; } const button = document.querySelector( '#withdrawAmount' ) ?.parentElement ?.querySelector('button'); showWithdrawMessage( "Submitting withdrawal request...", false ); try { const { data, error } = await supabaseClient.rpc( "request_withdrawal", { p_amount: amount, p_reference: phone } ); if (error) { console.error( "Withdrawal request error:", error ); showWithdrawMessage( error.message, true ); return; } const result = Array.isArray(data) ? data[0] : data; if ( !result || result.success !== true ) { showWithdrawMessage( "The withdrawal request could not be created.", true ); return; } document .getElementById("withdrawAmount") .value = ""; showWithdrawMessage( `Withdrawal request #${result.transaction_id} submitted successfully. It is pending administrator approval.`, false ); await loadTransactions(); } catch (error) { console.error( "Withdrawal request error:", error ); showWithdrawMessage( error.message || "Unable to submit withdrawal request.", true ); } }
